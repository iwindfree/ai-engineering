```python
# Python Program to Calculate the First 10,000 Terms of the Series
# 1 - 1/3 + 1/5 - 1/7 + ... multiplied by 4

total = 0.0
for i in range(10000):
    term = 1 / (2 * i + 1)
    if i % 2 == 0:
        total += term
    else:
        total -= term
result = total * 4
print(result)
```
